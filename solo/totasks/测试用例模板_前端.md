## Vue.js 组件测试模板

### 组件单元测试

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest'
import { mount } from '@vue/test-utils'
import TaskCard from '@/components/child/mission-card/index.vue'

describe('TaskCard 组件', () => {
  let wrapper

  beforeEach(() => {
    wrapper = mount(TaskCard, {
      props: {
        task: {
          id: 1,
          title: '完成作业',
          description: '数学和语文作业',
          status: 'pending',
          reward: 10,
          steps: ['第一步', '第二步', '第三步']
        }
      }
    })
  })

  it('应该正确渲染任务标题', () => {
    expect(wrapper.find('.task-title').text()).toBe('完成作业')
  })

  it('应该正确显示任务奖励', () => {
    expect(wrapper.find('.task-reward').text()).toContain('10')
  })

  it('应该在点击完成按钮时触发事件', async () => {
    const completeButton = wrapper.find('.complete-button')
    await completeButton.trigger('click')
    expect(wrapper.emitted('complete')).toHaveLength(1)
  })

  it('应该根据状态显示正确的样式', () => {
    expect(wrapper.classes()).toContain('status-pending')
  })
})
```

## 组合式函数 (Composable) 测试模板

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest'
import { useAudioPlayer } from '@/composables/useAudioPlayer'

describe('useAudioPlayer', () => {
  let audioPlayer

  beforeEach(() => {
    audioPlayer = useAudioPlayer()
  })

  it('应该初始化为暂停状态', () => {
    expect(audioPlayer.isPlaying.value).toBe(false)
  })

  it('应该能够播放音频', async () => {
    await audioPlayer.play('test-audio-url')
    expect(audioPlayer.isPlaying.value).toBe(true)
  })

  it('应该能够暂停音频', async () => {
    await audioPlayer.play('test-audio-url')
    await audioPlayer.pause()
    expect(audioPlayer.isPlaying.value).toBe(false)
  })

  it('应该能够设置音量', () => {
    audioPlayer.setVolume(50)
    expect(audioPlayer.volume.value).toBe(50)
  })

  it('音量应该在有效范围内', () => {
    audioPlayer.setVolume(150)
    expect(audioPlayer.volume.value).toBe(100)
    
    audioPlayer.setVolume(-10)
    expect(audioPlayer.volume.value).toBe(0)
  })
})
```

## API 服务测试模板

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest'
import { getDeviceList, updateDeviceConfig } from '@/services/device'
import { request } from '@/services/request'

vi.mock('@/services/request')

describe('Device API 服务', () => {
  beforeEach(() => {
    vi.clearAllMocks()
  })

  it('应该正确获取设备列表', async () => {
    const mockDevices = [
      { id: 1, name: '设备1', status: 'online' },
      { id: 2, name: '设备2', status: 'offline' }
    ]
    
    vi.mocked(request).mockResolvedValueOnce({
      code: 200,
      data: mockDevices
    })

    const result = await getDeviceList()
    
    expect(request).toHaveBeenCalledWith('/api/device/list')
    expect(result).toEqual(mockDevices)
  })

  it('应该正确更新设备配置', async () => {
    const configData = {
      id: 1,
      name: '新设备名',
      volume: 70
    }
    
    vi.mocked(request).mockResolvedValueOnce({
      code: 200,
      message: '更新成功'
    })

    const result = await updateDeviceConfig(configData)
    
    expect(request).toHaveBeenCalledWith('/api/device/update', {
      method: 'POST',
      data: configData
    })
  })

  it('应该在API错误时抛出异常', async () => {
    vi.mocked(request).mockRejectedValueOnce(new Error('网络错误'))

    await expect(getDeviceList()).rejects.toThrow('网络错误')
  })
})
```

## E2E 测试模板 (Playwright)

```typescript
import { test, expect } from '@playwright/test'

test.describe('登录流程 E2E 测试', () => {
  test('应该能够成功登录', async ({ page }) => {
    // 导航到登录页
    await page.goto('/login')

    // 填写登录表单
    await page.fill('input[name="username"]', 'testuser')
    await page.fill('input[name="password"]', 'password123')

    // 点击登录按钮
    await page.click('button[type="submit"]')

    // 验证是否跳转到首页
    await expect(page).toHaveURL('/home')
    await expect(page.locator('.welcome-message')).toContainText('欢迎回来')
  })

  test('应该在密码错误时显示错误', async ({ page }) => {
    await page.goto('/login')

    await page.fill('input[name="username"]', 'testuser')
    await page.fill('input[name="password"]', 'wrongpassword')
    await page.click('button[type="submit"]')

    await expect(page.locator('.error-message')).toContainText('用户名或密码错误')
  })
})

test.describe('设备管理 E2E 测试', () => {
  test('应该能够查看设备列表', async ({ page }) => {
    await page.goto('/device')
    
    await expect(page.locator('.device-list')).toBeVisible()
    const deviceCards = page.locator('.device-card')
    await expect(deviceCards).toHaveCountGreaterThan(0)
  })

  test('应该能够编辑设备配置', async ({ page }) => {
    await page.goto('/device')
    
    await page.click('.device-card:first-child .edit-button')
    await page.fill('input[name="deviceName"]', '更新后的设备名')
    await page.click('button.save-button')
    
    await expect(page.locator('.success-message')).toContainText('保存成功')
  })
})
```

## Store 测试模板

```typescript
import { describe, it, expect, beforeEach, vi } from 'vitest'
import { setActivePinia, createPinia } from 'pinia'
import { useTaskStore } from '@/store/task'

describe('Task Store', () => {
  beforeEach(() => {
    setActivePinia(createPinia())
  })

  it('应该初始化为空任务列表', () => {
    const store = useTaskStore()
    expect(store.tasks).toEqual([])
  })

  it('应该能够添加任务', () => {
    const store = useTaskStore()
    const newTask = {
      id: 1,
      title: '新任务',
      status: 'pending'
    }
    
    store.addTask(newTask)
    
    expect(store.tasks).toHaveLength(1)
    expect(store.tasks[0]).toEqual(newTask)
  })

  it('应该能够更新任务状态', () => {
    const store = useTaskStore()
    store.addTask({ id: 1, title: '任务1', status: 'pending' })
    
    store.updateTaskStatus(1, 'completed')
    
    expect(store.tasks[0].status).toBe('completed')
  })

  it('应该能够删除任务', () => {
    const store = useTaskStore()
    store.addTask({ id: 1, title: '任务1', status: 'pending' })
    
    store.deleteTask(1)
    
    expect(store.tasks).toEqual([])
  })

  it('应该能够计算已完成任务数量', () => {
    const store = useTaskStore()
    store.addTask({ id: 1, title: '任务1', status: 'completed' })
    store.addTask({ id: 2, title: '任务2', status: 'pending' })
    store.addTask({ id: 3, title: '任务3', status: 'completed' })
    
    expect(store.completedTasksCount).toBe(2)
  })
})
```
