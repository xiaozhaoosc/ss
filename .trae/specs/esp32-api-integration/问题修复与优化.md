### 3.1 ESP32 连接 MQTT Broker 失败 (P001)

**问题原因**：
- MQTT Broker 地址配置错误
- 网络连接不稳定
- 认证信息错误

**解决方案**：
1. 检查 MQTT Broker 地址配置
2. 实现网络状态检测，确保网络连接正常
3. 实现 MQTT 连接重连机制
4. 增加连接失败的错误处理和日志

**代码优化**：
```python
# mqtt_client.py 中的连接方法优化
def connect(self):
    """连接到 MQTT Broker"""
    max_retries = 5
    retry_count = 0
    while retry_count < max_retries:
        try:
            self.client = MQTTClient(
                client_id=self.client_id,
                server=self.server,
                port=self.port,
                user=self.user,
                password=self.password,
                keepalive=self.keepalive
            )
            self.client.connect()
            self.connected = True
            print('MQTT connected successfully')
            return True
        except Exception as e:
            print(f'MQTT connection failed (attempt {retry_count+1}/{max_retries}): {e}')
            retry_count += 1
            time.sleep(2 ** retry_count)  # 指数退避
    self.connected = False
    return False
```

```python
# network_manager.py 中的重连方法优化
def reconnect(self):
    """重新连接网络"""
    if self.is_connected():
        return True
    
    max_retries = 5
    retry_count = 0
    while retry_count < max_retries:
        print(f'Reconnecting to WiFi (attempt {retry_count+1}/{max_retries})...')
        if self.connect():
            return True
        retry_count += 1
        wait_time = min(30, 2 ** retry_count)
        print(f'Waiting {wait_time} seconds before next attempt...')
        time.sleep(wait_time)
    return False
```

```python
# mqtt_client.py 中的发布方法优化
def publish(self, topic, message, qos=1):
    """发布消息"""
    if not self.connected:
        if not self.connect():
            # 缓存消息到本地
            self._cache_message(topic, message)
            return False
    
    try:
        if isinstance(message, dict):
            message = json.dumps(message)
        self.client.publish(topic, message, qos=qos)
        return True
    except Exception as e:
        print('Publish error:', e)
        # 缓存消息到本地
        self._cache_message(topic, message)
        self.connected = False
        return False
```

class DataManager:

        self.cache_size = cache_size
    
    def _cache_message(self, topic, message):

        self.message_cache.append((topic, message))

private RedisTemplate<String, Object> redisTemplate;

public DeviceStatus getDeviceStatus(String deviceId) {
    // 先从缓存获取
    String key = "device:status:" + deviceId;
    DeviceStatus status = (DeviceStatus) redisTemplate.opsForValue().get(key);
    if (status != null) {
        return status;
    }

### 3.10 设备重启后配置丢失 (P010)

**问题原因**：
- 配置未持久化到 flash
- 设备重启后配置重置

**解决方案**：
1. 实现配置持久化，将配置保存到 flash
2. 设备启动时从 flash 加载配置
3. 增加配置备份机制

**代码优化**：
```python
# config_manager.py 中的配置持久化优化
import json
import os
import uos

class ConfigManager:
    def __init__(self, config_file='config.json'):
        self.config_file = config_file
        self.config = {}
        self.load_config()
    
    def load_config(self):
        """加载配置"""
        try:
            with open(self.config_file, 'r') as f:
                self.config = json.load(f)
        except Exception as e:
            print('Load config error:', e)
            # 使用默认配置
            self.config = {
                "wifi": {
                    "ssid": "",
                    "password": ""
                },
                "api": {
                    "url": "http://api.example.com"
                },
                "device": {
                    "id": "",
                    "type": "esp32",
                    "mac_address": "",
                    "firmware_version": "1.0.0"
                },
                "mqtt": {
                    "broker": "",
                    "client_id": "",
                    "username": "",
                    "password": ""
                }
            }
            self.save_config()
    
    def save_config(self):
        """保存配置"""
        try:
            # 确保目录存在
            if not os.path.exists(os.path.dirname(self.config_file)):
                os.makedirs(os.path.dirname(self.config_file))
            
            with open(self.config_file, 'w') as f:
                json.dump(self.config, f)
            
            # 同步到 flash
            uos.sync()
            return True
        except Exception as e:
            print('Save config error:', e)
            return False
```
